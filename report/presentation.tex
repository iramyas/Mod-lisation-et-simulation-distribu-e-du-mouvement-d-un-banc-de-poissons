\subsection{Les 3 règles de Reynolds}
% --- SLIDE 3 ---
\begin{frame}{Les 3 Règles de Reynolds}
\begin{enumerate}
    \setlength\itemsep{1em}
    \item \textbf{Séparation} : Éviter les collisions avec les voisins proches
    \item \textbf{Alignement} : S'adapter à la vélocité moyenne du groupe local
    \item \textbf{Cohésion} : Se diriger vers le centre de masse des voisins
\end{enumerate}
\vspace{1cm}
\centering
\textit{Appliquées localement $\Rightarrow$ comportement global complexe et réaliste}
\end{frame}

% --- SLIDE 4 ---
\begin{frame}{Formulation Mathématique}
Pour chaque boid $b$, on calcule trois forces :
\vspace{0.3cm}

\textbf{Séparation :}
$$\vec{F}_{sep} = -\sum_{i \in N(b)} \frac{\vec{p}_b - \vec{p}_i}{\|\vec{p}_b - \vec{p}_i\|^2}$$

\textbf{Alignement :}
$$\vec{F}_{ali} = \frac{1}{|N(b)|}\sum_{i \in N(b)} \vec{v}_i - \vec{v}_b$$

\textbf{Cohésion :}
$$\vec{F}_{coh} = \frac{1}{|N(b)|}\sum_{i \in N(b)} \vec{p}_i - \vec{p}_b$$

\vspace{0.2cm}
\small
Mise à jour : $\vec{a} = \vec{F}_{total}/m$, $\vec{v} \leftarrow \vec{v} + \vec{a}\Delta t$, $\vec{p} \leftarrow \vec{p} + \vec{v}\Delta t$
\end{frame}

% --- SLIDE 6  ---
\begin{frame}{Optimisation : SpatialGrid}

\vspace{0.5cm}

\textbf{\large 1. Le Problème} \\
Recherche de voisins naïve = $O(n^2)$ par frame. \textbf{Trop lent.}

\vspace{1cm}

\textbf{\large 2. La Solution : Partitionnement}
\begin{itemize}
    \setlength\itemsep{0.5em}
    \item Partitionner l'espace en cellules = $O(k)$ (où $k$ = boids/cellule)
    \item \textbf{Implémentation} : Grille 2D de vecteurs \texttt{Boid*}
    \item \textbf{Gain} : 500 boids $\rightarrow$ 5x plus rapide (50k comparaisons vs 250k)
\end{itemize}

\end{frame}

% --- SLIDE 7 ---
\begin{frame}{Benchmarks de performance}

\begin{itemize}
\setlength\itemsep{0.8em}

\item \textbf{Temps vs rayon (N = 500)} :
le SpatialGrid reste nettement plus rapide que l'algorithme naïf,
et l'écart se creuse lorsque le rayon de voisinage augmente.

\item \textbf{Temps vs nombre de boids (rayon = 50 px)} :
la courbe naïve suit une croissance quadratique en $N$,
tandis que le SpatialGrid se rapproche d'une croissance quasi linéaire.

\item Benchmarks obtenus en moyennant le temps par itération
sur plusieurs centaines de frames, afin de lisser les fluctuations
et comparer clairement les deux approches.
\end{itemize}

\section{Partie 2 - Perspectives version parallèle}
\subsection{Motivations}
% --- SLIDE 12 ---
\begin{frame}{Pourquoi paralléliser ?}
\begin{block}{1. Fidélité à la réalité}
    \begin{itemize}
        \item Dans la nature : chaque poisson agit \textbf{simultanément}
        \item Séquentiel : traitement un par un $\rightarrow$ s'éloigne de la réalité
        \item Parallèle/Distribué : calculs simultanés + synchronisation
    \end{itemize}
\end{block}

\begin{block}{2. Complexité algorithmique}
    \begin{itemize}
        \item Recherche de voisins : O(n * k) par frame (k = voisins/cellule)
        \item Pour n = 10000 boids à 60 FPS : calculs intensifs
        \item Limites du séquentiel : quelques centaines de boids max
    \end{itemize}
\end{block}
\end{frame}

\begin{frame}{Passage à l'échelle}
\begin{columns}
    \column{0.5\textwidth}
    \textbf{Limites actuelles :}
    \begin{itemize}
        \item $\sim$ 500 boids @ 60 FPS 
        \item Architecture séquentielle
        \item Calcul sur 1 coeur
    \end{itemize}

    \vspace{0.5cm}

    \textbf{Objectifs parallélisation :}
    \begin{itemize}
        \item \textbf{x20 plus d'agents} (10000 boids)
        \item Maintenir 60 FPS
        \item Exploiter architectures modernes
    \end{itemize}

    \column{0.5\textwidth}
    \centering
    \textbf{Speedup théorique* :}

    \vspace{0.3cm}

    \begin{tabular}{lc}
        \hline
        \textbf{Approche} & \textbf{Gain} \\
        \hline
        OpenMP (8 coeurs) & 5-6x \\
        MPI (4 noeuds x 8 coeurs) & 15-20x \\
    \end{tabular}

    \vspace{0.5cm}

    \footnotesize
    * Selon loi d'Amdahl avec 90\% du code parallélisable
\end{columns}
\end{frame}

\subsection{Stratégies de parallélisation}
\begin{frame}{Phase 1 : OpenMP (Mémoire Partagée)}
    \begin{block}{Pourquoi OpenMP ?}
        \begin{itemize}
            \item \textbf{Simplicité d'utilisation} : directives simple
            \item \textbf{Validation rapide} : tester scalabilité
        \end{itemize}
    \end{block}

    \begin{block}{Stratégie de parallélisation}
        \begin{enumerate}
            \item \textbf{Boucle principale}
            \begin{itemize}
                \item Indépendance : chaque boid lit les ses voisins
                \item Race condition : écriture position/vitesse 
            \end{itemize}
            \item \textbf{SpatialGrid}
            \begin{itemize}
                \item Défi : insertion concurrente
                \item Alternative : grille locale par thread  + merge
            \end{itemize}
        \end{enumerate}
    \end{block}
\end{frame}

\begin{frame}{Phase 2 : MPI (Mémoire Distribuée)}
    \textbf{Approche :} Distribution sur plusieurs machines/noeud
    \vspace{0.1cm}
    \begin{block}{Décomposition de domaine}
        \begin{itemize}
            \item Partitionnement spatial de la zone de simulation
            \item Chaque processus MPI gère une région
            \item Boids distribués selon leur position
        \end{itemize}
    \end{block}
    \begin{block}{Communication aux frontières}
        \begin{itemize}
            \item Echange des boids proches des frontières
            \item Synchronisation
            \item Migration des boids entre processus si changement de région
        \end{itemize}
    \end{block}
    \vspace{0.1cm}
    \small
    \textbf{Défi :} Equilibrage de charge + minimiser communications
\end{frame}

\begin{frame}{Optimisations et Défis}
\begin{enumerate}
    \item \textbf{Load balancing}
    \begin{itemize}
        \item \textbf{Problème} : Distribution inégale des boids
        \item \textbf{Détection} : Mesurer \# boids/processus chaque 100 frames
        \item \textbf{Action} : Repartitionnement si écart $>$ 30\%
        \item \textbf{Coût} : Communication vs gain calcul
    \end{itemize}
    
    \vspace{0.3cm}

    \item \textbf{Réduction Communications}
    \begin{itemize}
        \item \textbf{Ghost zone adaptative} : largeur = rayon\_max
        \item \textbf{Communication asynchrone} : Envoi et réception par MPI
        \item \textbf{Overlap} : calcul pendant transfert
        \item \textbf{Gain attendu} : 20-30\% temps communication
    \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Limitations et Défis Techniques}
    \begin{alertblock}{Défi identifiés}
        \begin{enumerate}
            \item \textbf{Overhead communication} (MPI)
            \begin{itemize}
                \item Latence réseau : 1-10 ms/message
                \item Impact si petit nombre de boids/processus
                \item \textbf{Seuil minimal} : $\sim$1000 boids pour rentabilité
            \end{itemize}
            \item \textbf{Scalabilité limitée}
            \begin{itemize}
                \item Loi d'Amdahl : partie séquentielle (init, render)
                \item \textbf{Efficacité} : 70-80\% attendue
            \end{itemize}

            \item \textbf{Load imbalance}
            \begin{itemize}
                \item Boids non uniformément répartis
                \item Repartitionnement coûteux
            \end{itemize}
        \end{enumerate}
    \end{alertblock}
\end{frame}

\begin{frame}{Métriques de Performance}
    \textbf{Métriques importantes :}
    \vspace{0.3cm}
    \begin{enumerate}
        \item \textbf{Speedup} : $S_p = \frac{T_1}{T_p}$ (temps séquentiel / temps parallèle)
        \item \textbf{Efficacité} : $E_p = \frac{S_p}{p}$ (speedup / nombre de processeurs)
        \item \textbf{Strong scaling} : Nombre de boid constant, augmenter le nombre de processeurs
        \item \textbf{Weak scaling} : Charge par processeur constante, augmenter la taille du problème
        \item \textbf{FPS maintenu} : Objectif 60 FPS avec 10000 boids
    \end{enumerate}
    \vspace{0.5cm}
    \small
    \textbf{Benchmarks comparatifs :} Séquentiel, OpenMP, MPI, Hybride
