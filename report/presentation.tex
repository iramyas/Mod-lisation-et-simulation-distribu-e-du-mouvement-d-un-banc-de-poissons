\section{Partie 1 - Version séquentielle}


\subsection{Contexte et objectifs}

\begin{frame}{Contexte}
  \begin{itemize}
    \item Modélisation d'un \textbf{système multi-agents} : un banc de poissons 2D.
    \item Aucun leader : chaque poisson suit uniquement ses voisins immédiats.
    \item Ordre global (\emph{flocking}) qui apparaît à partir de règles locales simples.
    \item Applications : finance, trafic, jeux vidéo, robotique, etc.
  \end{itemize}
  \vspace{0.4cm}
  \textbf{Référence} : modèle de flocking de Craig Reynolds (1987).
\end{frame}

\begin{frame}{Objectifs du semestre 1}
  \begin{itemize}
    \item Implémenter en C++ une simulation réaliste basée sur les \textbf{3 règles de Reynolds}.
    \item Vérifier que le comportement collectif émerge visuellement (banc cohérent).
    \item Atteindre le temps réel : viser \textbf{60 FPS} avec plusieurs centaines de boids.
    \item Préparer une base propre pour la version \textbf{parallèle / distribuée} du semestre 2.
  \end{itemize}
  \vspace{0.4cm}
  \textit{Environ 110 commits répartis entre Yohann, moi et Laurence.}
\end{frame}

\subsection{Les 3 règles de Reynolds}

\begin{frame}{Les 3 règles de Reynolds}
  \begin{enumerate}
    \setlength\itemsep{0.7em}
    \item \textbf{Séparation} : éviter les collisions avec les voisins trop proches.
    \item \textbf{Alignement} : s'adapter à la vitesse moyenne du groupe local.
    \item \textbf{Cohésion} : se diriger vers le centre de masse des voisins.
  \end{enumerate}
  \vspace{0.5cm}
  \centering
  \textit{Règles purement locales $\Rightarrow$ comportement global émergent.}
\end{frame}

\begin{frame}{Formulation mathématique des forces}
  Pour un boid $b$ de position $\vec{p}_b$ et de vitesse $\vec{v}_b$, avec ensemble de voisins $N(b)$ :

  \vspace{0.3cm}
  \textbf{Séparation (éviter les collisions)} :
  \[
    \vec{F}_{sep} = - \sum_{i \in N(b)} \frac{\vec{p}_b - \vec{p}_i}{\|\vec{p}_b - \vec{p}_i\|^2}
  \]

  \textbf{Alignement (synchroniser la direction)} :
  \[
    \vec{F}_{ali} = \frac{1}{|N(b)|}\sum_{i \in N(b)} \vec{v}_i - \vec{v}_b
  \]

  \textbf{Cohésion (rejoindre le groupe)} :
  \[
    \vec{F}_{coh} = \frac{1}{|N(b)|}\sum_{i \in N(b)} \vec{p}_i - \vec{p}_b
  \]

  \small Mise à jour : on applique la force totale, on limite par \texttt{maxForce}, puis on met à jour
  la vitesse (limitée par \texttt{maxSpeed}) et la position.
\end{frame}

\begin{frame}{Combinaison des 3 règles}
  À chaque frame, pour chaque boid :
  \begin{enumerate}
    \item On récupère ses voisins dans un \textbf{rayon de perception} (30–50 px suivant le paramètre).
    \item On calcule les trois forces : $\vec{F}_{sep}$, $\vec{F}_{ali}$, $\vec{F}_{coh}$.
    \item On les pondère par les poids de la GUI : $w_{sep}$, $w_{ali}$, $w_{coh}$.
    \item On applique la force résultante via \texttt{applyForce} puis \texttt{update}.
  \end{enumerate}

  %\vspace{0.1cm}
  \[
    \vec{F}_{total} =
    w_{sep}\,\vec{F}_{sep} +
    w_{ali}\,\vec{F}_{ali} +
    w_{coh}\,\vec{F}_{coh}
  \]

  \vspace{0.1cm}
  Les poids viennent directement des \textbf{sliders SFML} 
  %: on peut voir en temps réel
  %un banc très lâche (séparation forte) ou au contraire très compact (cohésion forte).
\end{frame}

\subsection{Architecture et implémentation}

\begin{frame}{Architecture logicielle (C++)}
  \begin{itemize}
    \item \textbf{Vector2D} : toutes les opérations vectorielles (position, vitesse, forces).
    \item \textbf{Boid} : état d’un agent + implémentation des règles (séparation, alignement, cohésion).
    \item \textbf{Flock} : contient le vecteur de \texttt{Boid} et une \texttt{SpatialGrid}.
    \item \textbf{SpatialGrid} : grille 2D de cellules stockant des pointeurs sur les boids.
    \item \textbf{GUI (SFML)} : création/suppression de boids, sliders pour les poids et le rayon, affichage.
  \end{itemize}
  \vspace{0.1cm}
  La méthode centrale est \texttt{Flock::updateAll(dt)} qui orchestre une itération complète de la simulation.
\end{frame}

  \textbf{Boid} :
  \begin{itemize}
    \item Attributs : \texttt{position}, \texttt{velocity}, \texttt{acceleration},
          \texttt{mass}, \texttt{maxSpeed = 30}, \texttt{maxForce = 2.5},

          \texttt{cohesion}, \texttt{seek}, \texttt{flee}, \texttt{update}.
  \end{itemize}

  \vspace{0.3cm}

    \item Méthodes : \texttt{populateRandom}, \texttt{setNeighborRadius}, \texttt{setWeights},   
    \\{updateAll}.
  \end{itemize}
\end{frame}

\subsection{Optimisation et benchmarks}

\begin{frame}{Limite de l'approche naïve}
  \textbf{Recherche de voisins sans grille} :
  \begin{itemize}
    \item Pour chaque boid, on parcourt \textbf{tous} les autres boids.
    \item Complexité : \(O(n^2)\) comparaisons par itération.
  \end{itemize}

  \vspace{0.4cm}
  \textbf{Conséquence pratique} (mesurée dans \texttt{benchmark.cpp}, sans affichage) :
  \begin{itemize}
    \item À rayon = 50 px :
    \begin{itemize}
      \item 100 boids : \(\approx 0{,}19\) ms par itération.
      \item 400 boids : \(\approx 2{,}84\) ms.
      \item 800 boids : \(\approx 10{,}33\) ms.
    \end{itemize}
    \item La courbe est clairement \textbf{quadratique} quand on augmente \(n\).
  \end{itemize}
\end{frame}

\begin{frame}{SpatialGrid : principe}
  \textbf{Idée} : réduire la recherche de voisins à l'environnement local.
  \begin{itemize}
    \item L'espace de simulation est découpé en cellules carrées de taille \texttt{cellSize}.
    \item Chaque cellule contient les pointeurs vers les boids qui s’y trouvent.
    \item Pour un boid donné, on ne regarde que sa cellule et les 8 voisines.
  \end{itemize}

  \vspace{0.4cm}
  \textbf{Complexité obtenue} :
  \begin{itemize}
    \item Insertion des boids dans la grille : \(O(n)\).
    \item Recherche des voisins : \(O(n \cdot k)\) avec \(k\) = nombre moyen de boids par cellule (petit).
    \item En pratique : proche d’un comportement \textbf{linéaire} en \(n\).
  \end{itemize}
\end{frame}

\begin{frame}{Benchmark 1 : coût vs rayon (N = 500)}

\section{Partie 2 - Perspectives version parallèle}
\subsection{Motivations}
% --- SLIDE 12 ---
\begin{frame}{Pourquoi paralléliser ?}
\begin{block}{1. Fidélité à la réalité}
    \begin{itemize}
        \item Dans la nature : chaque poisson agit \textbf{simultanément}
        \item Séquentiel : traitement un par un $\rightarrow$ s'éloigne de la réalité
        \item Parallèle/Distribué : calculs simultanés + synchronisation
    \end{itemize}
\end{block}

\begin{block}{2. Complexité algorithmique}
    \begin{itemize}
        \item Recherche de voisins : O(n * k) par frame (k = voisins/cellule)
        \item Pour n = 10000 boids à 60 FPS : calculs intensifs
        \item Limites du séquentiel : quelques centaines de boids max
    \end{itemize}
\end{block}
\end{frame}

\begin{frame}{Passage à l'échelle}
\begin{columns}
    \column{0.5\textwidth}
    \textbf{Limites actuelles :}
    \begin{itemize}
        \item $\sim$ 500 boids @ 60 FPS 
        \item Architecture séquentielle
        \item Calcul sur 1 coeur
    \end{itemize}

    \vspace{0.5cm}

    \textbf{Objectifs parallélisation :}
    \begin{itemize}
        \item \textbf{x20 plus d'agents} (10000 boids)
        \item Maintenir 60 FPS
        \item Exploiter architectures modernes
    \end{itemize}

    \column{0.5\textwidth}
    \centering
    \textbf{Speedup théorique* :}

    \vspace{0.3cm}

    \begin{tabular}{lc}
        \hline
        \textbf{Approche} & \textbf{Gain} \\
        \hline
        OpenMP (8 coeurs) & 5-6x \\
        MPI (4 noeuds x 8 coeurs) & 15-20x \\
    \end{tabular}

    \vspace{0.5cm}

    \footnotesize

\end{columns}
\end{frame}

\subsection{Stratégies de parallélisation}
\begin{frame}{Phase 1 : OpenMP (Mémoire Partagée)}
    \begin{block}{Pourquoi OpenMP ?}
        \begin{itemize}
            \item \textbf{Simplicité d'utilisation} : directives simples
            \item \textbf{Validation rapide} : tester scalabilité sur une macine
        \end{itemize}
    \end{block}

    \begin{block}{Stratégie de parallélisation}
        \begin{enumerate}
            \item \textbf{Boucle principale}
            \begin{itemize}
                \item Indépendance : chaque boid lit les ses voisins
                \item Race condition : écriture position/vitesse 
            \end{itemize}
            \item \textbf{SpatialGrid}
            \begin{itemize}
                \item Défi : insertion concurrente
                \item Alternative : grille locale par thread  + fusion finale
            \end{itemize}
        \end{enumerate}
    \end{block}
\end{frame}

\begin{frame}{Phase 2 : MPI (Mémoire Distribuée)}
    \textbf{Approche :} Distribution sur plusieurs machines
    \vspace{0.1cm}
    \begin{block}{Décomposition de domaine}
        \begin{itemize}
            \item Partitionnement spatial de la zone de simulation
            \item Chaque processus MPI gère une région
            \item Boids distribués selon leur position
        \end{itemize}
    \end{block}
    \begin{block}{Communication aux frontières}
        \begin{itemize}
            \item Echange des boids proches des frontières
            \item Synchronisation
            \item Migration des boids entre processus si changement de région
        \end{itemize}
    \end{block}
    \vspace{0.1cm}
    \small
    \textbf{Défi :} Equilibrage de charge + minimiser communications
\end{frame}

\begin{frame}{Phase 3 : Approche Hybride MPI + OpenMP}
  \begin{itemize}
    \item \textbf{MPI} : Gère la décomposition spatiale (grille distribuée)
    \begin{itemize}
      \item Chaque processus MPI = une région de l'espace
      \item Communication inter-processus pour les frontières
    \end{itemize}
    \item \textbf{OpenMP} : Parallélise les boids au sein de chaque région
    \begin{itemize}
      \item Threads multiples par processus MPI
      \item Traitement parallèle des boids locaux
    \end{itemize}
  \end{itemize}
  
\end{frame}

\begin{frame}{Optimisations et Défis}
\begin{enumerate}
    \item \textbf{Load balancing}
    \begin{itemize}
        \item \textbf{Problème} : Distribution inégale des boids
        \item \textbf{Détection} : Mesurer \# boids/processus chaque 100 frames
        \item \textbf{Action} : Repartitionnement si écart $>$ 30\%
        \item \textbf{Coût} : Communication vs gain calcul
    \end{itemize}
    
    \vspace{0.3cm}

    \item \textbf{Réduction Communications}
    \begin{itemize}
        \item \textbf{Ghost zone adaptative} : largeur = rayon\_max
        \item \textbf{Communication asynchrone} : Envoi et réception par MPI
        \item \textbf{Overlap} : calcul pendant transfert
        \item \textbf{Gain attendu} : 20-30\% temps communication
    \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Limitations et Défis Techniques}
    \begin{alertblock}{Défi identifiés}
        \begin{enumerate}
            \item \textbf{Overhead communication} (MPI)
            \begin{itemize}
                \item Latence réseau : 1-10 ms/message
                \item Impact si petit nombre de boids/processus
                \item \textbf{Seuil minimal} : $\sim$1000 boids pour rentabilité
            \end{itemize}
            \item \textbf{Scalabilité limitée}
            \begin{itemize}
                \item Loi d'Amdahl : partie séquentielle (init, render)
                \item \textbf{Efficacité} : 70-80\% attendue
            \end{itemize}

            \item \textbf{Load imbalance}
            \begin{itemize}
                \item Boids non uniformément répartis
                \item Repartitionnement coûteux
            \end{itemize}
        \end{enumerate}
    \end{alertblock}
\end{frame}

\begin{frame}{Métriques de Performance}
    \textbf{Métriques importantes :}
    \vspace{0.3cm}
    \begin{enumerate}
        \item \textbf{Speedup} : $S_p = \frac{T_1}{T_p}$ (temps séquentiel / temps parallèle)
        \item \textbf{Efficacité} : $E_p = \frac{S_p}{p}$ (speedup / nombre de processeurs)
        \item \textbf{Strong scaling} : Nombre de boid constant, augmenter le nombre de processeurs
        \item \textbf{Weak scaling} : Charge par processeur constante, augmenter la taille du problème
        \item \textbf{FPS maintenu} : Objectif 60 FPS avec 10000 boids
    \end{enumerate}
    \vspace{0.5cm}
    \small
    \textbf{Benchmarks comparatifs :} Séquentiel, OpenMP, MPI, Hybride
